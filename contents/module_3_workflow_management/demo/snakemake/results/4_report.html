We can additionally plot the data in a nice little graphic.

top_names <- read.csv("/home/rstudio/project/contents/module_3_workflow_management/demo/snakemake/work/3_top_names.csv")

top_names |>
  # The upper bound is deliberate: the data set gains a year of records over
  # time, and bounding the window keeps this figure matching its caption.
  group_by(sex, name) |>
  summarise(births = sum(births)) |>
  slice_max(order_by = births, n = 10) |>
  ggplot(aes(x = reorder(name, births), y = births)) +
  facet_wrap(~ sex, scales = "free") +
  geom_bar(stat = "identity") +
  labs(x = "Name", y = "Births", 
       title = "") +
  coord_flip() +
  theme_minimal()
`summarise()` has grouped output by 'sex'. You can override using the `.groups`
argument.

We can also plot my own name over time.

datclean <- read.csv("/home/rstudio/project/contents/module_3_workflow_management/demo/snakemake/work/3_selected_names.csv")

datclean |>
  group_by(name, year) |>
  filter(name %in% c("Christian", "Adrian")) |>
  ggplot(aes(x = year, y = births, color = name)) +
  geom_line() +
  geom_point() +
  scale_color_viridis_d() +
  labs(x = "Year", y = "Births", color = "") +
  expand_limits(y = 0) +
  theme_minimal()
